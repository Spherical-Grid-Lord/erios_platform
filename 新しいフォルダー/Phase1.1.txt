governance/provenance.py
from datetime import datetime
from datetime import UTC

from pydantic import Field

from erios_egp.core.base import EGPBaseModel
from erios_egp.core.enums import Layer


class Checksum(EGPBaseModel):

    algorithm: str = "sha256"

    value: str


class Provenance(EGPBaseModel):

    source_id: str

    source_type: str

    source_layer: Layer

    source_uri: str

    checksum: Checksum

    trusted: bool = True

    retrieved_at: datetime = Field(
        default_factory=lambda: datetime.now(UTC)
    )


class KnowledgeChainEntry(EGPBaseModel):

    source: str

    identifier: str

    timestamp: datetime = Field(
        default_factory=lambda: datetime.now(UTC)
    )


class ProvenanceChain(EGPBaseModel):

    entries: list[KnowledgeChainEntry] = Field(
        default_factory=list
    )

schemas/export.py
from pathlib import Path

from erios_egp.governance.envelope import (
    EGPEnvelope
)


def export_schema():

    schema = EGPEnvelope.model_json_schema()

    target = Path(
        "egp.schema.json"
    )

    target.write_text(
        __import__("json").dumps(
            schema,
            indent=2,
            ensure_ascii=False
        ),
        encoding="utf-8"
    )

    return target
